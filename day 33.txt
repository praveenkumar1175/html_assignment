Task 1 — Setup a New Project

Create a new HTML file and link Bootstrap via CDN.

Add a <div class="container"> and confirm styling works.


<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Bootstrap Day 33</title>

    <!-- Bootstrap CDN -->
    <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.8/dist/css/bootstrap.min.css" rel="stylesheet">
</head>
<body>

    <div class="container">
        <h1>Bootstrap Connected </h1>
        <p>If you see spacing and proper font, Bootstrap is working.</p>
    </div>

</body>
</html>


Task 2 — Typography & Spacing

Add headings, paragraphs, and apply text utilities like .text-center, .text-primary.

Use spacing utilities (p-, m-) to style sections.


<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Bootstrap Typography</title>

    <!-- Bootstrap CDN -->
    <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.8/dist/css/bootstrap.min.css" rel="stylesheet">
</head>
<body>

    <div class="container mt-5">

        <!-- Heading -->
        <h1 class="text-center text-primary mb-4">
            Welcome to Bootstrap
        </h1>

        <!-- Paragraph -->
        <p class="text-center text-secondary mb-3">
            This is a simple paragraph using Bootstrap text utilities.
        </p>

        <!-- Another Section -->
        <div class="p-4 bg-light mt-4">
            <h2 class="text-success">Spacing Example</h2>
            <p class="m-2">
                This section uses padding (p-4) and margin (m-2).
            </p>
        </div>

    </div>

</body>
</html>


.container → Centers content + adds margin
.text-center → Align text center
.text-primary → Blue color
p-4 → Padding
m-2 → Margin
mt-5 / mb-4 → Margin top / bottom
